// Trace of matrix to daiagonal elements:

#include <stdio.h>

int main() {
    int size;

    printf("Enter the matrix size: ");
    scanf("%d", &size);

    int matrix[size][size]; 

    printf("Enter the elements: \n");
    for (int i = 0; i < size; i++) {
        for (int k = 0; k < size; k++) {
            scanf("%d", &matrix[i][k]);
        }
    }

    int trace = 0; 
    for (int i = 0; i < size; i++) {
        trace += matrix[i][i];
    }

    printf("The trace of the matrix: %d\n", trace);
    return 0;
}
