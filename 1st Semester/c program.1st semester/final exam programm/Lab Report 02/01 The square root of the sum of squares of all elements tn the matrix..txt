//The square root of the sum of squares of all elements in the matrix.

#include <stdio.h>
#include <math.h>

int main()
{
    int size;
    printf("Enter the matrix size: ");
    scanf("%d",&size);
    
    int arr[size][size];
    for(int i=0; i<size; i++){
        for(int k=0; k<size; k++){
            scanf("%d",&arr[i][k]);
        }
    }
    
    //The square root of the sum of squares of all elements in the matrix:
    int sum;
    for(int i=0; i<size; i++){
        for(int k=0; k<size; k++){
            
            sum += arr[i][k]*arr[i][k];
        }
    }
    double normal= sqrt(sum);
    printf("The normal of the matrix: %.2f\n", normal);
    return 0;
}
