#include<stdio.h>

void addstring(char str1[], char str2[]){
    int i,k;
    
    for(i=0; str1[i] !='\0'; i++);
    
    for(k=0; str2[k] !='\0'; k++){
        str1[i+k] = str2[k];
    }
}

int main(){
    char str1[50];
    printf("Enter the string: ");
    scanf("%s",&str1);
    
    char str2[50];
    printf("Enter the seceond string: ");
    scanf("%s",&str2);
    
    
    addstring(str1,str2);
    printf("Concatenate string: %s\n ", str1);
    return 0;
}
