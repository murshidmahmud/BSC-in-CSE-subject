#include<stdio.h>

int check_LowerTriangular(int matrix[][60], int size){
    for(int i=0; i<size; i++){
        for(int k=i+1; k<size; k++){
            if(matrix[i][k] !=0){
                return 0;
            }
        }
    }
    return 1;
}
int main(){
    int size;
    printf("Enter the matrix size: ");
    scanf("%d",&size);
    
    int matrix[60][60];
    printf("Enter the element: \n");
    for(int i=0; i<size; i++){
        for(int k=0;k<size; k++){
            scanf("%d",&matrix[i][k]);
        }
    }
    
    if(check_LowerTriangular(matrix,size)){
        printf("The matrix is a lower traingular matrix.\n");
    }
    else{
        printf("The matrix isn't lower traingular matrix.");
    }
    return 0;
}
