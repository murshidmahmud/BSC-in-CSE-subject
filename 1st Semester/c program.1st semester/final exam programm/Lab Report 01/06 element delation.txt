// Delete elements in an array:


#include <stdio.h>

int main() {
   int size;
   printf("Enter your array size: ");
   scanf("%d", &size);
   
   int arr[size];
   
   for(int i = 0; i< size; i++)
   {
       printf("%d index: ", i);
       scanf("%d", &arr[i]);
   }
   
   //printf("Array elements is: ");
   //for(int i = 0; i<size; i++)
   //{
   //    printf("%d\t", arr[i]);
  // }
  
   // Delete elements in an array:
   
   int index;
   printf("\nEnter the index to delete: ");
   scanf("%d", &index);
   
   for(int i = index; i < size - 1; i++)
   {
       arr[i] = arr[i+1];
   }
   
   size--;
   
   printf("Array after deletion: ");
   
   for(int i = 0; i < size; i++)
   {
       printf("%d\t", arr[i]);
   }
   
   

    return 0;
}
