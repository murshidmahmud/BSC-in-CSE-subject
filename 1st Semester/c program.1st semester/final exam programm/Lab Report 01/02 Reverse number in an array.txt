// find the reverse number in an array:


#include <stdio.h>

int main() {
   int size;
   printf("Enter your array size: ");
   scanf("%d", &size);
   
   int arr[size];
   for(int i = 0; i< size; i++)
   {
       printf("%d index: ", i);
       scanf("%d", &arr[i]);
   }
   
   //printf("Array elements is: ");
   //for(int i = 0; i<size; i++)
   //{
   //    printf("%d\t", arr[i]);
  // }
  

   // Reverse an array
   printf("\nElements of reverse array: ");
   for(int i = size - 1; i >= 0; i--)
   {
       printf("%d\t", arr[i]);
   }
   

    return 0;
}
