#include<stdio.h>

int main(){
    
    int size;
    printf("Enter the number: ");
    scanf("%d",&size);
    
    int arr[size];
    for(int i=0; i<size; i++){
        printf("%d index: ", i);
        scanf("%d", &arr[i]);
    }

    //Array position element insertion: 
    
    int pos,n;
    printf("Choose the position: ");
    scanf("%d",&pos);
    
    printf("Choose the element: ");
    scanf("%d", &n);
    
    for(int i=size; i>pos ; i--){
        
        arr[i] = arr[i-1];
    }
    arr[pos] = n;
    size++;
    
    printf("Show the new array: ");
    for(int i=0; i<size; i++){
        
        printf("%d\t", arr[i]);
    }
    return 0;
}
