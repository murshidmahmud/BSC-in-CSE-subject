//Linear Search:

#include<stdio.h>

int binarySearch(int arr[], int size, int target){
    
    int low =0, high = size-1;
    
    while(low<=high){
        int mid = (low+high)/2;
        
        if(arr[mid] == target){
            return mid;
        }
        else if(arr[mid]<target){
            low= mid+1;
        }
        else{
            high = mid-1;
        }
    }
    return -1;
}

int main() {
    
    int size,target;
    
    printf("Enter the array size: ");
    scanf("%d", &size);
    
    int arr[size];
    
    printf("Enter array value: \n");
    for(int i=0; i<size; i++){
        printf("%d index: ", i);
        scanf("%d", &arr[i]);
    }
    
    printf("Array is: ");
    for(int i=0; i<size; i++){
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    printf("Enter search element: ");
    scanf("%d", &target);
    
    int rslt = binarySearch(arr,size,target);
    
    if(rslt !=-1){
        printf("Element %d found at index %d\n", target,rslt);
    }
    else{
        printf("Element %d not found in the array\n", target);
    }
}
