//Linear Search:

#include<stdio.h>

int linearSearch(int arr[], int size, int target){
    
    for(int i =0; i<size; i++){
        if(arr[i] == target){
            return i;
        }
    }
    return -1;
}

int main() {
    
    int size,target;
    
    printf("Enter the array size: ");
    scanf("%d", &size);
    
    int arr[size];
    
    printf("Enter array value: \n");
    for(int i=0; i<size; i++){
        printf("%d index: ", i);
        scanf("%d", &arr[i]);
    }
    
    printf("Array is: ");
    for(int i=0; i<size; i++){
        printf("%d ", arr[i]);
    }
    printf("\n");
    
    printf("Enter search element: ");
    scanf("%d", &target);
    
    int rslt = linearSearch(arr,size,target);
    
    if(rslt !=-1){
        printf("Element %d found at index %d\n", target,rslt);
    }
    else{
        printf("Element %d not found in the array\n", target);
    }
}
